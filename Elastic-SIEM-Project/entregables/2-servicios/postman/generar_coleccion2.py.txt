contrato, tipos y regla de negocio. La colección trae además las tres aserciones globales.

Los casos CP-SERV-61, 63 y 65 esperan 400. Hoy el sistema devuelve 500 (defectos D1 a D4): el valor
esperado NO se cambia para que pasen.

Se guarda como `.py.txt` para que ni ruff ni la prueba de no-autonomía lo recorran. Uso:
    python3 entregables/2-servicios/postman/generar_coleccion2.py.txt
"""

# ─── Construcción de scripts ────────────────────────────────────────────────

def cabecera(cp: str, objetivo: str, requisitos: str, tecnica: str, peticion: str, esperado: str) -> str:
    def campo(nombre: str, texto: str) -> list[str]:
        return textwrap.wrap(f"{nombre}: {texto}", width=68, initial_indent="// ",
                             subsequent_indent="// ") or [f"// {nombre}:"]
    lineas = [RAYA, f"// {cp} | {objetivo}"]
    for nombre, texto in (("Requisitos", requisitos), ("Técnica", tecnica),
                          ("Petición", peticion), ("Esperado", esperado)):
        lineas += campo(nombre, texto)
    return "\n".join(lineas + [RAYA])


def cuatro(cp: str, status: int, contrato: tuple[str, str], tipos: str, regla: tuple[str, str],
           *, json_: bool = True, extra: str = "") -> str:
    """Las cuatro aserciones: 1 código, 2 contrato, 3 tipos, 4 regla de negocio."""
    partes = [f'''// 1. Código: {status}.
pm.test("{cp} | Código {status}", function () {{
    pm.response.to.have.status({status});
}});
''']
    if json_:
        partes.append("const datos = pm.response.json();\n")
    else:
        partes.append("const cuerpo = pm.response.text();\n")
    partes.append(f'''// 2. Contrato: {contrato[0]}.
pm.test("{cp} | Contrato: {contrato[0]}", function () {{
{_sangrar(contrato[1])}
}});
''')
    partes.append(f'''// 3. Tipos de datos.
pm.test("{cp} | Tipos de datos", function () {{
{_sangrar(tipos)}
}});
''')
    partes.append(f'''// 4. Regla de negocio: {regla[0]}.
pm.test("{cp} | {regla[0]}", function () {{
{_sangrar(regla[1])}
}});''')
    if extra:
        partes.append("\n" + extra.strip("\n"))
    return "\n".join(partes)


def _sangrar(js: str) -> str:
    return "\n".join("    " + l if l.strip() else l for l in js.strip("\n").split("\n"))


def error_js(esperado_texto: str | None = None) -> tuple[tuple[str, str], str]:
    contrato = ("el error trae el campo error", 'pm.expect(datos).to.have.property("error");')
    tipos = 'pm.expect(datos.error).to.be.a("string").and.not.empty;'
    return contrato, tipos


# ─── Peticiones ─────────────────────────────────────────────────────────────

def pedido(nombre: str, metodo: str, url: str, script: str, *, token: bool | str = True,
           cuerpo: str | None = None, previo: str | None = None, seguir_redirecciones: bool = True) -> dict:
    cab = []
    if token is True:
        cab.append({"key": "X-Internal-Token", "value": "{{token}}", "type": "text"})
    elif token == "falso":
        cab.append({"key": "X-Internal-Token", "value": "token-falso", "type": "text"})
    if cuerpo is not None:
        cab.append({"key": "Content-Type", "value": "application/json", "type": "text"})
    sin_esquema = url.split("{{", 1)[1].split("}}", 1)
    host = "{{" + sin_esquema[0] + "}}"
    ruta = sin_esquema[1]
    path_ = [p for p in ruta.split("?")[0].split("/") if p]
    objeto_url = {"raw": url, "host": [host], "path": path_}
    if "?" in ruta:
        objeto_url["query"] = [{"key": k, "value": v} for k, v in
                               (par.split("=", 1) for par in ruta.split("?", 1)[1].split("&"))]
    req = {"method": metodo, "header": cab, "url": objeto_url}
    if cuerpo is not None:
        req["body"] = {"mode": "raw", "raw": cuerpo, "options": {"raw": {"language": "json"}}}
    eventos = []
    if previo:
        eventos.append({"listen": "prerequest",
                        "script": {"type": "text/javascript", "exec": previo.split("\n")}})
    eventos.append({"listen": "test", "script": {"type": "text/javascript", "exec": script.split("\n")}})
    item = {"name": nombre, "event": eventos, "request": req}
    if not seguir_redirecciones:
        item["protocolProfileBehavior"] = {"followRedirects": False}
    return item


def caso(cp: str, objetivo: str, requisitos: str, tecnica: str, peticion: str, esperado: str,
         metodo: str, url: str, status: int, contrato: tuple[str, str], tipos: str,
         regla: tuple[str, str], **kw) -> dict:
    extra = kw.pop("extra", "")
    json_ = kw.pop("json_", True)
    script = cabecera(cp, objetivo, requisitos, tecnica, peticion, esperado) + "\n\n" + cuatro(
        cp, status, contrato, tipos, regla, json_=json_, extra=extra)
    return pedido(f"{cp} | {objetivo}", metodo, url, script, **kw)


def preparacion(nombre: str, metodo: str, url: str, para: str, status: int, extra: str = "",
                cuerpo: str | None = None) -> dict:
    """Pedido auxiliar (no es un caso del plan): deja listo lo que necesita un caso posterior."""
    script = (cabecera(nombre, "preparación", "—", "No es un caso de prueba", f"{metodo} {url}", para)
              + f'\n\n// Verifica que la preparación salió bien.\npm.test("{nombre} | Código {status}", '
                f'function () {{\n    pm.response.to.have.status({status});\n}});\n' + extra)
    return pedido(nombre, metodo, url, script, token=False, cuerpo=cuerpo)
